\section{Work Plan: Phase 1 --- Benchmark Games and Metrics}
%==========================================================================

\textit{Estimated time: 2--3 hours coding + 1 hour running.}

\subsection{Task 1.1: Add canonical benchmark games}

Implement named game constructors in \texttt{game.py}:

\begin{enumerate}[nosep]
    \item \textbf{Prisoner's Dilemma} (2 players, 2 actions):
    \[
    u_1 = \begin{pmatrix} -1 & -3 \\ 0 & -2 \end{pmatrix}, \quad
    u_2 = \begin{pmatrix} -1 & 0 \\ -3 & -2 \end{pmatrix}
    \]

    \item \textbf{Shapley Game} (2 players, 3 actions): the classic game where best-response dynamics cycle but CE exists.
    \[
    u_1 = \begin{pmatrix} 1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 1 \end{pmatrix}, \quad
    u_2 = \begin{pmatrix} 0 & 1 & 0 \\ 0 & 0 & 1 \\ 1 & 0 & 0 \end{pmatrix}
    \]

    \item \textbf{Matching Pennies} (2 players, 2 actions): zero-sum.

    \item \textbf{Chicken / Hawk-Dove} (2 players, 2 actions): multiple NE, non-trivial CE.

    \item \textbf{Theorem~5 counterexample} (2 players, $4 \times 2$): already in tests, expose as named constructor.
\end{enumerate}

\subsection{Task 1.2: Add $\varepsilon$-CE gap metric (NashConv)}

The $\varepsilon$-CE gap is the maximum incentive to deviate from the recommendation:
\[
\varepsilon\text{-CE}(x) \;=\; \max_{i \in [n]} \;\max_{a_i, a_i' \in A_i} \;\phi_i(a_i, a_i', x)
\]
This is the standard metric in the no-regret $\to$ CE literature. Implement as a method on \texttt{NormalFormGame} and track it per-round in \texttt{CuttingPlaneAlgorithm}.

Add to \texttt{CuttingPlaneResult}: field \texttt{ce\_gap: NDArray} of shape $(T,)$.

\subsection{Task 1.3: Add baselines}

Add to \texttt{baselines.py}:

\begin{enumerate}[nosep]
    \item \textbf{Random recommendation:} $x(t) = \text{Uniform}(\Delta(A))$ every round. No learning. Lower bound.

    \item \textbf{Oracle:} $x(t) \in \CE(w^\star)$ (true game known). Zero compliance regret after first round. Upper bound.

    \item \textbf{Regret matching} (Hart--Mas-Colell): each player $i$ maintains regret vector $R_i(a_i, a_i')$ and plays proportional to positive regrets. Standard CE convergence baseline. Converges at $\bigO(\sqrt{T})$.

    \item \textbf{EXP3 per player:} bandit baseline, each player learns independently. Already implemented (MW baseline). Rename for clarity.
\end{enumerate}

\section{Work Plan: Phase 2 --- Main Paper Experiments (4 Figures)}
%==========================================================================

\textit{Estimated time: 3--4 hours running (parallelizable).}

\subsection{Figure 1: Compliance Regret vs $T$ (Theorem 8)}

\textbf{Goal:} Show $R(T) = \bigO(nM \log T)$.

\begin{itemize}[nosep]
    \item \textbf{Panel A:} Regret vs $t$ on log-log scale. Curves for Prisoner's Dilemma ($nM{=}8$), Shapley ($nM{=}18$), $(2,2,2)$ ($nM{=}24$), $(3,3,3)$ ($nM{=}81$), $(2,)^{\times 5}$ ($nM{=}160$).
    \item \textbf{Panel B:} $R(t) / \log(t)$ vs $t$. Should plateau at constant $\propto nM$.
    \item \textbf{Reference lines:} $c \cdot nM \cdot \log(t)$ fitted per curve.
    \item Games: 5 games (3 canonical + 2 random). $T = 5{,}000$. Seeds: 30. Feedback: BR ($\gamma{=}0$).
    \item \textbf{Baselines on same plot:} Random recommendation (linear regret), Oracle (zero).
\end{itemize}

\subsection{Figure 2: Learnability Separation --- QR vs BR (Theorems 4 \& 5)}

\textbf{Goal:} QR converges to equivalence class, BR does not.

\begin{itemize}[nosep]
    \item \textbf{Panel A (QR):} Distance to equivalence class $d(w(t), [w^\star])$ vs $t$. Should $\to 0$.
    \item \textbf{Panel B (BR):} Same metric. Should $\to$ positive constant.
    \item \textbf{Panel C (Theorem 5 counterexample):} On game $u$ with BR, plot $d(w(t), [w^\star_u])$ and $d(w(t), [w^\star_v])$. Both small (indistinguishable). With QR, only $d(w(t), [w^\star_u]) \to 0$.
    \item Games: Shapley $(3,3)$, $(2,)^{\times 5}$, counterexample $(4,2)$. $T = 5{,}000$. Seeds: 30.
\end{itemize}

\subsection{Figure 3: Moderator vs No-Moderator --- Swap Regret}

\textbf{Goal:} Moderator gives $\bigO(\log T)$ swap regret, MW gives $\bigO(\sqrt{T})$.

\begin{itemize}[nosep]
    \item \textbf{Log-log plot.} $x$-axis: $t$ (log). $y$-axis: avg player swap regret (log).
    \item \textbf{Curves:} Moderator ($\gamma$-BR), MW baseline, Regret Matching baseline.
    \item \textbf{Reference slopes:} $c_1 \log(t)$ and $c_2 \sqrt{t}$.
    \item Games: Shapley, $(2,)^{\times 5}$, $(3,3,3)$. $T = 10{,}000$. Seeds: 30.
    \item $\gamma = 0.01$ for the moderator.
\end{itemize}

\subsection{Figure 4: $\varepsilon$-CE Gap Convergence}

\textbf{Goal:} Show the recommended distribution converges to a CE of the true game.

\begin{itemize}[nosep]
    \item $x$-axis: $t$. $y$-axis: $\varepsilon$-CE gap of $x(t)$ w.r.t.\ true utilities.
    \item \textbf{Curves:} Moderator (cutting-plane), Regret Matching, Random.
    \item Games: Shapley, Prisoner's Dilemma, $(2,)^{\times 5}$. $T = 5{,}000$. Seeds: 30.
    \item \textbf{Expected:} Moderator $\to 0$ fast (once CE learned, gap is zero). Regret Matching $\to 0$ at $\bigO(1/\sqrt{T})$. Random stays high.
\end{itemize}

\section{Work Plan: Phase 4 --- Compact Pipeline for Congestion Games}
%==========================================================================

\textit{Estimated time: 1--2 days coding + testing.}

\subsection{Task 4.1: Implement congestion game environment}
\begin{itemize}[nosep]
    \item Network $G = (V, E)$ with source-destination pairs.
    \item Affine edge costs: $c_e(\ell) = \theta_{e,1} + \theta_{e,2} \ell$.
    \item Route enumeration (all simple $s_i$--$t_i$ paths up to length bound).
    \item Feature vectors $\phi_i(a)$ and difference features $\Delta\phi_i(r, r', a_{-i})$.
    \item Parameter dimension $d = 2|E|$.
\end{itemize}

\subsection{Task 4.2: Implement compact CE solver}
\begin{itemize}[nosep]
    \item Product-distribution CCE: variables $f_{i,r}$ (marginals per agent per route).
    \item LP with $\sum_i m_i$ variables instead of $M = \prod_i m_i$.
    \item Incentive constraints using expected congestion moments.
\end{itemize}

\subsection{Task 4.3: Implement compact oracle}
\begin{itemize}[nosep]
    \item Cut directly in $\R^d$: $q_\theta = \Phi^\top q$ computed via expected congestion.
    \item No enumeration of opponent profiles.
    \item BR/QR feedback using marginals.
\end{itemize}

\subsection{Task 4.4: Implement compact cutting-plane}
\begin{itemize}[nosep]
    \item Reuse ACCPM in $\R^d$ (already implemented, just change dimension).
    \item Per-round: $\theta(t) \to$ reconstruct costs $\to$ compact CE $\to$ sample $\to$ observe $\to$ cut in $\R^d$.
\end{itemize}
